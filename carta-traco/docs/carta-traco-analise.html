// domain/mcc.js
export function calcularMCC(materiais, custos, unidade) {
  const itens = materiais.map(m => {
    const c = custos.find(c => c.unidade === unidade &&
      c.material.toLowerCase() === m.produto.toLowerCase() &&
      (m.fornecedor === '-' || c.fornecedor === m.fornecedor));
    const unit = c ? converterCustoParaKg(+c.valorLiquido, c.unidadeMedida, m.massaEspecifica) : 0;
    return { ...m, custoUnitario: unit, custoTotal: m.quantidade * unit, semCusto: !c };
  });
  return { itens, total: itens.reduce((s,i)=>s+i.custoTotal,0), pendencias: itens.filter(i=>i.semCusto) };
}
D · SQL PostgreSQL proposto 
CREATE TABLE ct_carta (
  id BIGSERIAL PRIMARY KEY, nome TEXT NOT NULL, revisao INT NOT NULL, unidade TEXT NOT NULL,
  status TEXT NOT NULL CHECK (status IN ('Ativa','Inativa')), vigencia DATE NOT NULL,
  observacao TEXT, criado_por TEXT NOT NULL, criado_em TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (nome, unidade, revisao));
CREATE UNIQUE INDEX ux_ct_carta_ativa ON ct_carta (nome, unidade) WHERE status = 'Ativa';

CREATE TABLE ct_carta_item (
  carta_id BIGINT REFERENCES ct_carta(id) ON DELETE CASCADE, material TEXT NOT NULL, fornecedor TEXT,
  qtd_kg NUMERIC(10,3) NOT NULL CHECK (qtd_kg >= 0), massa_esp NUMERIC(5,3) NOT NULL,
  PRIMARY KEY (carta_id, material));

CREATE TABLE ct_custo (
  id BIGSERIAL PRIMARY KEY, unidade TEXT NOT NULL, fornecedor TEXT NOT NULL, material TEXT NOT NULL,
  un_medida TEXT NOT NULL, valor NUMERIC(12,4) NOT NULL, icms_pct NUMERIC(5,2) NOT NULL DEFAULT 0,
  frete NUMERIC(12,4) NOT NULL DEFAULT 0, tipo_frete TEXT CHECK (tipo_frete IN ('CIF','FOB')),
  valor_liquido NUMERIC(12,4) GENERATED ALWAYS AS (valor*(1-icms_pct/100)+frete) STORED,
  vigente_desde DATE NOT NULL DEFAULT current_date);
CREATE VIEW ct_custo_vigente AS SELECT DISTINCT ON (unidade,fornecedor,material) *
  FROM ct_custo ORDER BY unidade,fornecedor,material,vigente_desde DESC,id DESC;

function calcularIndicadores(materiais) {
  var cimento = 0, adicao = 0, agua = 0, aditivo1 = 0, aditivo2 = 0;
  var areias = [], britas = [];
  var totalAreias = 0, totalBritas = 0;

  materiais.forEach(function(m) {
    if (m.produto.startsWith('Cimento')) cimento += m.quantidade;
    else if (m.produto === 'Adição') adicao += m.quantidade;
    else if (m.grupo === 'agua') agua += m.quantidade;
    else if (m.produto === 'Aditivo 1') aditivo1 += m.quantidade;
    else if (m.produto === 'Aditivo 2') aditivo2 += m.quantidade;
    if (m.grupo === 'areia') { areias.push(m); totalAreias += m.quantidade; }
    if (m.grupo === 'brita') { britas.push(m); totalBritas += m.quantidade; }
  });

  var pesoTotal = materiais.reduce(function(s, m) { return s + m.quantidade; }, 0);
  var volume = materiais.reduce(function(s, m) { return s + (m.quantidade / m.massaEspecifica); }, 0);
  var aglomerante = cimento + adicao;
  var relacaoAC = cimento > 0 ? agua / cimento : 0;

  // Taxa de argamassa = miúdos / (miúdos + graúdos) — sem água
  var totalAgregados = totalAreias + totalBritas;
  var taxaArgamassa = totalAgregados > 0 ? (totalAreias / totalAgregados) * 100 : 0;

  // Taxa de argamassa em volume
  var volAreias = 0, volBritas = 0;
  areias.forEach(function(a) { volAreias += a.quantidade / a.massaEspecifica; });
  britas.forEach(function(b) { volBritas += b.quantidade / b.massaEspecifica; });
  var totalVolAgregados = volAreias + volBritas;
  var taxaArgamassaVol = totalVolAgregados > 0 ? (volAreias / totalVolAgregados) * 100 : 0;

  var pctAditivo1 = cimento > 0 ? (aditivo1 / cimento) * 100 : 0;
  var pctAditivo2 = cimento > 0 ? (aditivo2 / cimento) * 100 : 0;

  var distAreias = areias.map(function(a) {
    return { nome: a.produto, quantidade: a.quantidade, pct: totalAreias > 0 ? (a.quantidade / totalAreias) * 100 : 0 };
  });
  var distBritas = britas.map(function(b) {
    return { nome: b.produto, quantidade: b.quantidade, pct: totalBritas > 0 ? (b.quantidade / totalBritas) * 100 : 0 };
  });

  return {
    pesoTotal: pesoTotal, volume: volume, relacaoAC: relacaoAC,
    taxaArgamassa: taxaArgamassa, taxaArgamassaVol: taxaArgamassaVol,
    pctAditivo1: pctAditivo1, pctAditivo2: pctAditivo2,
    distAreias: distAreias, distBritas: distBritas, totalAreias: totalAreias, totalBritas: totalBritas
  };
}

// ========== CONVERSÃO DE CUSTO PARA KG ==========
function converterCustoParaKg(valor, unidadeMedida, massaEspecifica) {
  if (!valor || valor === 0) return 0;
  switch ((unidadeMedida || 'kg').toLowerCase()) {
    case 'kg': return valor;
    case 'ton': return valor / 1000;
    case 'litro': return valor * (massaEspecifica || 1);
    case 'm³': case 'm3': return valor / ((massaEspecifica || 1) * 1000);
    case 'saco': return valor / 50;
    default: return valor;
  }
}

async function calcularMCC() {
  var materiais = getMateriaisFromForm();
  if (materiais.length === 0) { showToast('Preencha pelo menos um material com quantidade.', 'error'); return; }

  var unidadeCarta = document.getElementById('carta-unidade').value;
  if (!unidadeCarta) { showToast('Selecione a unidade de vigência antes de calcular.', 'error'); return; }

  var ind = calcularIndicadores(materiais);
  document.getElementById('indicadores-section').style.display = 'block';
  renderIndicadores(ind);

  showLoading();
  try {
    var result = await apiGet('getCustos', { unidade: unidadeCarta });
    if (result.error) throw new Error(result.error);
    var custos = result.data || [];

    var mccData = [];
    var mccTotal = 0;

    materiais.forEach(function(mat) {
      var custoItem = custos.find(function(c) {
        var matchMaterial = c.material.toLowerCase() === mat.produto.toLowerCase() && c.unidade === unidadeCarta;
        if (mat.fornecedor && mat.fornecedor !== '-') return matchMaterial && c.fornecedor === mat.fornecedor;
        return matchMaterial;
      });
      var custoUnitario = 0;
      if (custoItem) {
        custoUnitario = converterCustoParaKg(
          parseFloat(custoItem.valorLiquido),
          custoItem.unidadeMedida || 'kg',
          mat.massaEspecifica
        );
      }
      var custoTotal = mat.quantidade * custoUnitario;
      mccTotal += custoTotal;
      mccData.push({
        produto: mat.produto, fornecedor: custoItem ? custoItem.fornecedor : '-',
        quantidade: mat.quantidade, massaEspecifica: mat.massaEspecifica,
        unidade: 'kg', custoUnitario: custoUnitario, custoTotal: custoTotal
      });
    });

    document.getElementById('mcc-preview-section').style.display = 'block';
    document.getElementById('mcc-total-value').textContent = formatCurrency(mccTotal);
    renderMCCTable(mccData, mccTotal, 'mcc-table-container');
    renderPieChart(mccData, mccTotal, 'chart-cadastro-mcc', 'cadastro');

    if (mccData.some(function(m) { return m.custoUnitario === 0; })) {
      showToast('Atenção: Alguns materiais não possuem custo cadastrado.', 'error');
    } else {
      showToast('MCC e indicadores calculados!', 'success');
    }
  } catch (error) {
    showToast('Erro ao calcular MCC: ' + error.message, 'error');
  } finally {
    hideLoading();
  }
}

function renderPieChart(mccData, mccTotal, canvasId, chartKey) {
  var canvas = document.getElementById(canvasId);
  if (!canvas) return;
  if (chartKey === 'cadastro' && chartCadastro) chartCadastro.destroy();
  else if (chartKey === 'detalhe' && chartDetalhe) chartDetalhe.destroy();

  var labels = mccData.map(function(m) { return m.produto; });
  var data = mccData.map(function(m) { return m.custoTotal; });
  var colors = generateColors(mccData.length);

  var chart = new Chart(canvas, {
    type: 'pie',
    data: { labels: labels, datasets: [{ data: data, backgroundColor: colors, borderWidth: 2, borderColor: '#fff' }] },
    options: {
      responsive: true, maintainAspectRatio: true,
      plugins: {
        legend: { position: 'bottom', labels: { font: { size: 11 }, padding: 12 } },
        tooltip: { callbacks: { label: function(ctx) {
          var pct = mccTotal > 0 ? ((ctx.parsed / mccTotal) * 100).toFixed(1) : 0;
          return ctx.label + ': ' + formatCurrency(ctx.parsed) + ' (' + pct + '%)';
        }}}
      }
    }
  });
  if (chartKey === 'cadastro') chartCadastro = chart;
  else chartDetalhe = chart;
}

function generateColors(count) {
  var palette = ['#3498db','#e74c3c','#2ecc71','#f39c12','#9b59b6','#1abc9c','#e67e22','#34495e','#16a085','#c0392b','#8e44ad','#27ae60','#d35400','#2c3e50','#f1c40f','#7f8c8d','#2980b9','#e84393','#00b894','#fdcb6e'];
  var colors = [];
  for (var i = 0; i < count; i++) colors.push(palette[i % palette.length]);
  return colors;
}

async function salvarCarta() {
  var nome = document.getElementById('carta-nome').value.trim();
  var revisao = parseInt(document.getElementById('carta-revisao').value) || 0;
  var unidade = document.getElementById('carta-unidade').value;
  var vigencia = document.getElementById('carta-vigencia').value;
  var status = document.getElementById('carta-status').value;
  var observacao = document.getElementById('carta-observacao').value.trim();
  var materiais = getMateriaisFromForm();

  if (!nome || !unidade || !vigencia) { showToast('Preencha todos os campos obrigatórios.', 'error'); return; }
  if (materiais.length === 0) { showToast('Informe pelo menos um material.', 'error'); return; }

  showLoading();
  try {
    var result = await apiPost('salvarCarta', {
      nome: nome, revisao: revisao, unidade: unidade, vigencia: vigencia, status: status,
      observacao: observacao,
      materiais: materiais.map(function(m) {
        return { produto: m.produto, quantidade: m.quantidade, massaEspecifica: m.massaEspecifica, unidade: m.unidade, fornecedor: m.fornecedor };
      }),
      dataCadastro: getToday()
    });
    if (result.error) throw new Error(result.error);
    showToast('Carta Traço salva com sucesso!', 'success');
    limparCadastro();
  } catch (error) {
    showToast('Erro ao salvar: ' + error.message, 'error');
  } finally {
    hideLoading();
  }
}

function renderCartasList(cartas) {
  var container = document.getElementById('consulta-results');
  if (cartas.length === 0) {
    container.innerHTML = '<p class="empty-state">Nenhuma carta traço encontrada.</p>';
    return;
  }

  var grupos = {};
  cartas.forEach(function(c) {
    var key = c.nome + '|' + c.unidade + '|' + c.revisao;
    if (!grupos[key]) {
      grupos[key] = { id: c.id, nome: c.nome, revisao: c.revisao, unidade: c.unidade,
        status: c.status, vigencia: c.vigencia, dataCadastro: c.dataCadastro,
        tipoFrete: c.tipoFrete || '', observacao: c.observacao || '', materiais: [] };
    }
    grupos[key].materiais.push({
      fornecedor: c.fornecedor, produto: c.produto,
      quantidade: c.quantidade, massaEspecifica: c.massaEspecifica || 0, unidadeMedida: c.unidadeMedida
    });
  });

  var html = '';
  Object.values(grupos).forEach(function(carta) {
    var isInativa = carta.status === 'Inativa';
    html += '<div class="carta-card ' + (isInativa ? 'inativa' : '') + '"' +
      " onclick='abrirDetalhe(" + JSON.stringify(carta).replace(/'/g, "&#39;") + ")'>" +
      '<div class="carta-card-header">' +
      '<span class="carta-card-title">' + escapeHtml(carta.nome) + '</span>' +
      '<span class="carta-card-badge ' + (isInativa ? 'badge-inativa' : 'badge-ativa') + '">' + carta.status + '</span></div>' +
      '<div class="carta-card-info">' +
      '<span>\ud83d\udccb Rev. ' + carta.revisao + '</span>' +
      '<span>\ud83c\udfed ' + escapeHtml(carta.unidade) + '</span>' +
      '<span>\ud83d\udcc5 ' + formatDate(carta.vigencia) + '</span>' +
      '<span>\ud83d\udce6 ' + carta.materiais.length + ' materiais</span></div></div>';
  });
  container.innerHTML = html;
}

async function editarCarta() {

  document.getElementById('carta-unidade').value = carta.unidade || '';

  document.getElementById('carta-status').value = 'Ativa';
  document.getElementById('carta-observacao').value = carta.observacao || '';

  await carregarFornecedores();

  // Limpar todos os campos de materiais primeiro
  MATERIAIS_TRACO.forEach(function(mat) {
    var qtdEl = document.getElementById('mat-' + mat.id);
    var meEl = document.getElementById('me-' + mat.id);
    var fornEl = document.getElementById('forn-' + mat.id);
    if (qtdEl) qtdEl.value = '';
    if (meEl) meEl.value = mat.meDefault.toFixed(3);
    if (fornEl) fornEl.value = '';
  });

  // Preencher materiais da carta
  (carta.materiais || []).forEach(function(m) {
    var isCimento = (m.produto || '').toLowerCase().startsWith('cimento');
    var def = MATERIAIS_TRACO.find(function(d) {
      if (isCimento) return d.id === 'cimento';
      return d.nome.toLowerCase() === (m.produto || '').toLowerCase();
    });
    if (def) {
      if (isCimento) {
        document.getElementById('tipo-cimento').value = m.produto || 'Cimento CP V';
      }
      var qtdEl = document.getElementById('mat-' + def.id);
      var meEl = document.getElementById('me-' + def.id);

      if (meEl) meEl.value = parseFloat(m.massaEspecifica || def.meDefault).toFixed(3);

function getCartas(params) {
  const sheet = getSheet('cadastro');
  const data = sheet.getDataRange().getValues();

  if (data.length <= 1) {
    return { data: [] };
  }

  const headers = data[0];
  const rows = data.slice(1);

  let cartas = rows.map(row => ({
    id: row[0],
    nome: row[1],
    revisao: row[2],
    unidade: row[3],
    status: row[4],
    vigencia: formatDateForJson(row[5]),
    dataCadastro: formatDateForJson(row[6]),
    fornecedor: row[7],
    produto: row[8],

    unidadeMedida: row[10],
    massaEspecifica: row[11] || 0,

  }));

  if (params.unidade && params.unidade !== '') {

  }

  // Filtrar inativas
  if (params.mostrarInativas !== 'true') {

function salvarCarta(payload) {
  const sheet = getSheet('cadastro');
  const cartaId = generateId();
  const nome = payload.nome;
  const revisao = parseInt(payload.revisao) || 0;
  const unidade = payload.unidade;
  const vigencia = payload.vigencia;
  const status = payload.status || 'Ativa';
  const dataCadastro = payload.dataCadastro || Utilities.formatDate(new Date(), 'America/Sao_Paulo', 'yyyy-MM-dd');
  const observacao = payload.observacao || '';
  const tipoFrete = payload.tipoFrete || '';
  const materiais = payload.materiais || [];

  // Se a nova carta é Ativa, inativar versões anteriores do mesmo nome e unidade
  if (status === 'Ativa') {
    inativarVersaoAnterior(sheet, nome, unidade);
  }

  // Salvar cada material como uma linha
  materiais.forEach(mat => {
    sheet.appendRow([
      cartaId,
      nome,
      revisao,
      unidade,
      status,
      vigencia,
      dataCadastro,
      mat.fornecedor,
      mat.produto,
      mat.quantidade,
      mat.unidade,
      mat.massaEspecifica || 0,
      observacao,
      tipoFrete
    ]);
  });

  // Atualizar aba Resumo
  atualizarResumo();

  return { success: true, id: cartaId, message: 'Carta Traço salva com sucesso!' };
}

function inativarVersaoAnterior(sheet, nome, unidade) {
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return;

  for (let i = 1; i < data.length; i++) {
    const rowNome = data[i][1];
    const rowUnidade = data[i][3];
    const rowStatus = data[i][4];

    if (rowNome === nome && rowUnidade === unidade && rowStatus === 'Ativa') {
      // Coluna E (index 5 na planilha, mas col 5) = Status
      sheet.getRange(i + 1, 5).setValue('Inativa');
    }
  }
}

  // Ordenar: mais recente primeiro (para rastreamento)
  custos.reverse();

  return { data: custos };
}

function atualizarResumo() {
  const cadastroSheet = getSheet('cadastro');
  const custosSheet = getSheet('custos');
  const resumoSheet = getSheet('resumo');

  const cadastroData = cadastroSheet.getDataRange().getValues();
  const custosData = custosSheet.getDataRange().getValues();

  if (cadastroData.length <= 1) return;

  // Obter todos os custos indexados por material+unidade (último valor)
  const custosMap = {};
  if (custosData.length > 1) {
    custosData.slice(1).forEach(row => {
      const key = (row[4] + '|' + row[2]).toLowerCase(); // material|unidade
      custosMap[key] = parseFloat(row[9]) || 0; // valor líquido
    });
  }

  // Agrupar cartas por nome+unidade, pegar somente a de maior revisão
  const cartasMap = {};
  cadastroData.slice(1).forEach(row => {
    const id = row[0];
    const nome = row[1];
    const revisao = parseInt(row[2]) || 0;
    const unidade = row[3];
    const status = row[4];
    const vigencia = row[5];
    const produto = row[8];
    const quantidade = parseFloat(row[9]) || 0;

    const key = nome + '|' + unidade;

    if (!cartasMap[key]) {
      cartasMap[key] = {
        nome: nome,
        unidade: unidade,
        revisao: revisao,
        status: status,
        vigencia: vigencia,
        materiais: []
      };
    }

    // Se encontramos uma revisão maior, substituir
    if (revisao > cartasMap[key].revisao) {
      cartasMap[key].revisao = revisao;
      cartasMap[key].status = status;
      cartasMap[key].vigencia = vigencia;
      cartasMap[key].materiais = [];
    }

    // Adicionar material se for da revisão mais recente
    if (revisao === cartasMap[key].revisao) {
      cartasMap[key].materiais.push({ produto, quantidade, unidade: unidade });
    }
  });

  // Calcular MCC para cada carta
  const resumoRows = [];
  Object.values(cartasMap).forEach(carta => {
    let mccTotal = 0;
    carta.materiais.forEach(mat => {
      const key = (mat.produto + '|' + carta.unidade).toLowerCase();
      const custoUnit = custosMap[key] || 0;
      mccTotal += mat.quantidade * custoUnit;
    });

    resumoRows.push([
      carta.unidade,
      carta.nome,
      carta.revisao,
      mccTotal,
      carta.status,
      carta.vigencia
    ]);
  });

  // Limpar aba resumo e reescrever
  resumoSheet.clear();
  initSheetHeaders(resumoSheet, 'resumo');

  if (resumoRows.length > 0) {
    resumoSheet.getRange(2, 1, resumoRows.length, 6).setValues(resumoRows);

    // Formatar coluna de MCC como moeda
    resumoSheet.getRange(2, 4, resumoRows.length, 1)
      .setNumberFormat('R$ #,##0.00');
  }
}

// ========== UTILITY FUNCTIONS ==========

function formatDateForJson(date) {
  if (!date) return '';
  if (date instanceof Date) {
    return Utilities.formatDate(date, 'America/Sao_Paulo', 'yyyy-MM-dd');
  }
  return String(date);
}
